f-stringsSession 1

artist, plays, total = "Glass Tram", 229, 1175.0912
print(f"{artist} has {plays} plays")      # Glass Tram has 229 plays
print(f"next year: {plays + 1}")          # any expression fits in { }

:.1f one decimal, :, thousands separator, :12 pad to 12 wide, :>5 right-align in 5.

for, range and the accumulatorSession 2

for i in range(1, 6):        # 1 2 3 4 5   the end is NOT included
    print(i, end=" ")
print()

minutes = [4.65, 2.42, 7.10, 3.05]
total = 0                    # 1. set up BEFORE the loop
for m in minutes:
    total += m               # 2. update inside
print(round(total, 2))       # 3. use after: 17.22

range(5) is 0 to 4. range(0, 20, 5) is 0, 5, 10, 15. Put total = 0 inside the loop and it resets every time round.

while, break, continueSession 2

count = 3
while count > 0:
    print(count)
    count -= 1               # without this it never ends
print("Go")

for i in range(1, 100):
    if i * i > 50:
        print(i, "squared passes 50")   # 8 squared passes 50
        break                # stop the loop

for i in range(1, 11):
    if i % 2 == 0:
        continue             # skip to the next one
    print(i, end=" ")        # 1 3 5 7 9
print()

Functions

def and returnSession 3

def average(numbers):
    """Return the mean of a list of numbers."""
    return sum(numbers) / len(numbers)


result = average([4.65, 2.42, 7.10, 3.05])
print(round(result, 2))          # 4.3
print(round(average([2, 4]) + 1, 1))   # 4.0: a returned value can be used

Define once, call with brackets. Without them, average is the function itself and nothing runs. The docstring is what help(average) shows.

return versus printSession 3

def double_p(n):
    print(n * 2)          # shows a human, hands back nothing


def double_r(n):
    return n * 2          # hands a value back to the program


x = double_p(5)           # prints 10
print(x)                  # None
print(double_r(5) + 1)    # 11

double_p(5) + 1 is a TypeError about NoneType. A function with no return, or one that falls off the end of an if, gives None. Calculate and return; let the outer layer print.

Parameters, defaults, guard clausesSession 3

def describe(artist, plays, unit="plays"):
    if plays < 0:                          # guard clause: odd cases first
        return "not a real count"
    return f"{artist}: {plays} {unit}"


print(describe("DJ Kompas", 215))                 # DJ Kompas: 215 plays
print(describe("DJ Kompas", 1169.2, "minutes"))   # DJ Kompas: 1169.2 minutes
print(describe(plays=-1, artist="x"))             # by name, any order

List comprehensionsSession 4

minutes = [4.65, 2.42, 9.30, 0.51, 7.10]
long_ones = [m for m in minutes if m > 5]     # filter
print(long_ones)                              # [9.3, 7.1]
rounded = [round(m) for m in minutes]         # transform
print(rounded)                                # [5, 2, 9, 1, 7]
print(sum(1 for m in minutes if m < 1))       # 1: count matches

The same as a for loop with an append, on one line.

The counting patternSession 4

genres = ["Folk", "Jazz", "Folk", "Rock", "Folk"]
counts = {}
for g in genres:
    counts[g] = counts.get(g, 0) + 1
print(counts)                        # {'Folk': 3, 'Jazz': 1, 'Rock': 1}

# biggest first: keep this recipe
for g, n in sorted(counts.items(), key=lambda pair: pair[1], reverse=True):
    print(f"{g:6} {n:>3}")

A list of dictionaries is a table: one dictionary per row, one key per column. This loop is GROUP BY in session 6 and value_counts() in session 8.

Write a CSVSession 4

import os

jazz = [row for row in rows if row["genre"] == "Jazz"]
print(len(jazz))                         # 154

os.makedirs("output", exist_ok=True)     # "w" fails if the folder is missing
with open("output/jazz.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.DictWriter(f, fieldnames=rows[0].keys())
    writer.writeheader()
    writer.writerows(jazz)

"w" replaces the file without asking. newline="" stops blank lines on Windows. output/ is in .gitignore.

When things go wrong

try / exceptSession 12

def to_number(text):
    """Return text as a float, or None if it is not a number."""
    try:
        return float(text)
    except ValueError:               # name the error you expect
        return None


print(to_number("4.2"), to_number("4.2 min"), to_number(""))   # 4.2 None None

try:
    average([])
except ZeroDivisionError as error:
    print("could not average:", error)     # division by zero

Never a bare except:. It catches typos too, so a NameError turns into a quiet wrong answer. Catch a missing file in main(), not inside the loader.

Tests with pytestSession 12

# test_statistics_functions.py: a test is a function starting test_
import pytest


def test_average_of_two_numbers():
    assert average([2, 4]) == 3


def test_average_of_an_empty_list_raises():
    with pytest.raises(ZeroDivisionError):

Error messages, decodedSession 2 Session 3 Session 4

ErrorUsually means
SyntaxErrorA missing colon, bracket or quote, or = where == was meant. Smart quotes pasted from a chat app
IndentationErrorThe lines inside an if, loop or function are not indented the same
NameErrorA typo, a variable used before it exists, or one that only lived inside a function
TypeErrorMixing types: text plus a number, or a None from a function that printed instead of returning
ValueErrorRight type, wrong content: int("forty one")
KeyErrorA dictionary key or column name that is not there. Print .keys()
IndexErrorA position past the end of a list
FileNotFoundErrorRunning from the wrong folder. Print os.getcwd()
ZeroDivisionErrorDividing by len() of an empty list

Read the last line first: it names the error. Then the line number above it says where.
