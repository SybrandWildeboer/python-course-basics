Loading and looking

LoadSession 8

import sqlite3
import pandas as pd

plays = pd.read_csv("data/clean/plays.csv")

con = sqlite3.connect("data/music.db")
artists = pd.read_sql("SELECT * FROM artists", con)
awards = pd.read_sql("SELECT * FROM awards", con)
db_plays = pd.read_sql("SELECT * FROM plays", con)   # artist_id, no names
con.close()

read_sql takes any query, so the database can do the grouping and send back only the answer.

Filtering and sorting

MasksSession 8

long_ones = plays["minutes_played"] > 9    # one True/False per row
print(long_ones.sum())                     # 32: True counts as 1
print(len(plays[long_ones]))               # 32 rows kept

speaker_2025 = (plays["device"] == "speaker") & (plays["played_at"] >= "2025-01-01")
print(speaker_2025.sum())                  # 174

print(plays["device"].isin(["car", "tablet"]).sum())   # 284
print((~plays["device"].isin(["phone"])).sum())        # 1144: ~ is NOT

& and, | or, ~ not, and brackets round every condition. Python's and gives ValueError: The truth value of a Series is ambiguous.

The bracket trapSession 8

right = (plays["minutes_played"] > 5) & ((plays["device"] == "phone") | (plays["device"] == "laptop"))
wrong = (plays["minutes_played"] > 5) & (plays["device"] == "phone") | (plays["device"] == "laptop")
print(right.sum(), wrong.sum())     # 523 871

& is worked out before |, so the wrong one is "long phone plays, or any laptop play". It runs. Name each condition on its own line first.

Grouping and joining

value_counts and groupbySession 8

print(plays["genre"].value_counts().head(3))     # Electronic 547, Folk 438, Pop 302

print(plays.groupby("genre")["minutes_played"].sum().round(1).head(3))   # Afrobeat 596.8 ...

per_genre = plays.groupby("genre").agg(
    plays=("play_id", "count"),
    minutes=("minutes_played", "sum"),
    avg_min=("minutes_played", "mean"),
)
print(per_genre.sort_values("avg_min", ascending=False).round(2).head(2))  # Jazz 6.57 on top
print(per_genre[per_genre["plays"] > 300].index.tolist())   # HAVING: a mask after

agg takes new_name=("column", "function"). .reset_index() turns the group labels back into an ordinary column.

merge is JOINSession 8

named = db_plays.merge(artists, on="artist_id")             # inner join
print(len(db_plays), len(named))                            # 2183 2183: good

everyone = artists.merge(db_plays, on="artist_id", how="left")   # LEFT JOIN
print(len(everyone))                                             # 2185
print(everyone[everyone["play_id"].isna()]["artist_name"].tolist())
# ['Iron Fernway', 'Gravel Choir']

counts = everyone.groupby("artist_name")["play_id"].count()   # COUNT(p.play_id)
sizes = everyone.groupby("artist_name").size()                # COUNT(*)
print(counts["Iron Fernway"], sizes["Iron Fernway"])          # 0 1

merge is an inner join unless you say how="left". Count the rows before and after, every time.

Fan-out, and validate=Session 8

with_awards = db_plays.merge(awards, on="artist_id")
print(len(with_awards), round(with_awards["minutes_played"].sum(), 1))   # 2371 9732.1

per_artist = (db_plays.groupby("artist_id")
              .agg(plays=("play_id", "count"), minutes=("minutes_played", "sum"))
              .reset_index())                          # aggregate FIRST
n_awards = awards.groupby("artist_id").size().rename("awards").reset_index()
summary = per_artist.merge(n_awards, on="artist_id", how="left", validate="one_to_one")
summary["awards"] = summary["awards"].fillna(0).astype(int)   # COALESCE(x, 0)

try:
    db_plays.merge(awards, on="artist_id", validate="many_to_one")
except pd.errors.MergeError as error:
    print(error)   # Merge keys are not unique in right dataset; not a many-to-one merge

validate= turns a silent fan-out into an error. SQL has no such switch. drop_duplicates() is not the fix.

Cleaning

Find the damage firstSession 9

df = pd.read_csv("data/messy/plays_messy.csv")
print(df.shape)                       # (2223, 9): 40 more than the clean file
print(df.columns.tolist()[-1])        # 'minutes played ' with two spaces
print(df.isna().sum()[lambda s: s > 0].to_dict())
# {'genre': 36, 'country': 49, 'device': 70, 'minutes played ': 94}
print(df["device"].nunique())         # 20 spellings of five devices
print(df["device"].value_counts(dropna=False))   # NaN 70 shows up only with dropna=False
print(df.duplicated().sum())          # 40 exact duplicate rows

value_counts() and groupby hide missing values unless you pass dropna=False. Count the missing values before and after every fix.

Names, text and numbersSession 9

df = df.rename(columns={"minutes played ": "minutes_played"})
df["device"] = df["device"].str.strip().str.lower()    # ' PHONE' -> 'phone'
df["genre"] = df["genre"].str.strip().str.title()      # 'INDIE' -> 'Indie'

text = df["minutes_played"].str.replace(" min", "")    # '4.2 min' -> '4.2'
df["minutes_played"] = pd.to_numeric(text, errors="coerce")
print(df["minutes_played"].isna().sum())               # 94: only the real blanks

errors="coerce" turns anything unreadable into NaN without a word. Converting before stripping " min" loses 190 values instead of 94.

DatesSession 9

dates = df["played_at"]
parsed = pd.to_datetime(dates, format="%Y-%m-%d", errors="coerce")
parsed = parsed.fillna(pd.to_datetime(dates, format="%d/%m/%Y", errors="coerce"))
parsed = parsed.fillna(pd.to_datetime(dates, format="%d-%m-%Y", errors="coerce"))
print(parsed.isna().sum())                 # 0: every date read
df["played_at"] = parsed.dt.strftime("%Y-%m-%d")   # back to text, like the clean file

One format at a time, on purpose. format="mixed" raises no error and gets 159 dates wrong: it reads 03/09/2025 as the 9th of March.

Missing values and duplicatesSession 9

lookup = df.dropna(subset=["genre"]).groupby("artist_name")["genre"].first()
df["genre"] = df["genre"].fillna(df["artist_name"].map(lookup))   # fill from the artist

df = df.drop_duplicates()
print(len(df))                                  # 2183
df["device"] = df["device"].fillna("unknown")   # still counted, still visible

m = df["minutes_played"]
print(round(m.mean(), 3), round(m.fillna(0).mean(), 3))   # 4.121 3.943

.mean() skips NaN. fillna(0) claims those plays lasted zero minutes. dropna() with no arguments drops a row missing anything; say which column with subset=.
